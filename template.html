```javascript function calculateBasket(basketItems, networks) { let totalSum = 0; let missing_items = 0; // Отсортируем сети от дешевых к дорогим const sortedNetworks = networks.sort((a, b) => a.price - b.price); for (const item of basketItems) { let found = false; let price = 0; for (const network of sortedNetworks) { if (network.products.includes(item.id)) { price = network.price * item.quantity; totalSum += price; found = true; break; } } if (!found) { missing_items++; } } const lang = { en: 'price for', he: 'מחיר עבור' }; const message = `${lang[window.navigator.language]}` + ` ${totalSum.toFixed(2)} from ${basketItems.length - missing_items} of ${basketItems.length}`; console.log(message); } // Пример использования const basketItems = [ { id: 1, quantity: 3 }, { id: 2, quantity: 2 }, { id: 3, quantity: 1 } ]; const networks = [ { id: 1, price: 10, products: [1, 4] }, { id: 2, price: 20, products: [2, 5] }, { id: 3, price: 15, products: [3, 6] } ]; calculateBasket(basketItems, networks);
